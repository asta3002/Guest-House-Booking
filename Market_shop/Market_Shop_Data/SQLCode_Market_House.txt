create trigger ins_trans before insert on Transactions
for each row 
begin  

 declare Amnt bigint ;
 declare Pend bigint ;
 
 select Val into Amnt from Payments where Payments.Payment_Id = new.Payment_Id;
  if exists (select Pending_Charges.Payment_Id from Pending_Charges where Pending_Charges.Payment_Id = new.Payment_Id) then
       begin
       select Pending_Charges.Pending_Amount into Pend from Pending_Charges where Pending_Charges.Payment_Id = new.Payment_Id;
       if(Pend>new.Amount) then
       update Pending_Charges set Pending_Charges.Pending_Amount = Pending_Charges.Pending_Amount - new.Amount where Pending_Charges.Payment_Id = new.Payment_Id;
       elseif(Pend = new.Amount) then
       delete from Pending_Charges where Pending_Charges.Payment_Id = new.Payment_Id;
       end if;
       end;
         
  elseif (Amnt > new.Amount) then
    
   insert into Pending_Charges values(new.Payment_Id,(Amnt-new.Amount));


  
end if; 
end;
|

create trigger del_trans before delete on Transactions 
for each row
begin
 
 declare Pend bigint;
 if exists (select Pending_Charges.Payment_Id from Pending_Charges where Pending_Charges.Payment_Id = old.Payment_Id) then
        update Pending_Charges set Pending_Charges.Pending_Amount = Pending_Charges.Pending_Amount + old.Amount where Pending_Charges.Payment_Id = old.Payment_Id;

         
 else 
   insert into Pending_Charges values(old.Payment_Id,old.Amount);

end if;
end;
|


create procedure Update_Pendings()
begin
   declare done int default 0;
   declare P_Id varchar(10);
   declare Sm bigint default 0;
   declare P_amnt bigint default 0;
   declare P_cursor cursor for select Payment_Id from Payments;
   declare continue handler for not found set done =1;
   open P_cursor;
   Label: LOOP
     fetch P_cursor into P_Id;
     
     if done =1 then 
       leave Label;
     else
           
           if (not exists (select Payment_Id from Pending_Charges where Payment_ID = P_Id) ) then
               select Val into P_amnt from Payments where Payment_Id = P_Id;
               insert into pending_charges values (P_Id,P_amnt-Sm);
           end if;    
            
        
       end if;
       end LOOP;
       close P_cursor;
end;
|

create function Due_Amnt(Id varchar(10))
returns bigint 
deterministic
begin
 
 declare Pending bigint default 0;
  call Update_Pendings();
 select  sum(Pending_Charges.Pending_Amount) into Pending from( payments inner join pending_charges on pending_charges.Payment_ID = payments.Payment_ID) group by(Payments.Shop_Id) having payments.Shop_Id =Id;
 return Pending;
end;
|




create procedure Pending_Summary()
begin
   call Update_Pendings();
   select Payments.Shop_Id, sum(Pending_Charges.Pending_Amount) from (Pending_Charges inner join Payments on Payments.Payment_Id  = Pending_Charges.Payment_Id) group by(Payments.Shop_Id);
end;

create procedure Check_Pass(IN Id varchar(255) )
begin
    declare C_Date date;
   
    declare MsgA, MsgB, MsgC varchar(30);
    declare diff int ;
    set C_Date = CURDATE();
   
    set diff = DATEDIFF( C_Date, (SELECT SecurityPass.DateofExpiry from ( SecurityPass inner join ShopKpr on ShopKpr.Security_Id =SecurityPass.Security_Id) where ShopKpr.ShopKpr_Id = Id));
    set MsgA = "Pass Expired";
    set MsgB = "Valid Pass";

    if(diff>0) then
       select MsgA;
    else
       select MsgB;
    end if;
 end;
|

create procedure ShpKpr_Details(IN Id varchar(255))
begin 
 
   select ShopKpr.ShopKpr_Id, ShopKpr.Name,ShopKpr.Age,ShopKpr.Address,ShopKpr.Security_Id, SecurityPass.DateofExpiry as PassExpiryDate from (SecurityPass inner join  ShopKpr on ShopKpr.Security_Id =SecurityPass.Security_Id)  where(ShopKpr.ShopKpr_Id= Id);
   
end;
|

create procedure Expire_Pass() 
begin
        declare done int default 0; 
        declare S_ID varchar(255) ; 
        declare S_Iss, S_Exp date;
         declare C_Date date; 
        
        declare Pcursor CURSOR 
        for select * from SecurityPass ; 
        declare continue handler for not found set done = 1; 
        open Pcursor;
        set C_Date = CURDATE();
        
        label: LOOP 
        fetch Pcursor into S_ID,S_Iss,S_Exp;
        if done = 1 then 
        leave label;        
        else
           
           if ((SELECT DATEDIFF( S_Exp,C_Date)) <=5) then
              select S_ID, ShopKpr.ShopKpr_Id,ShopKpr.Name from (SecurityPass inner join ShopKpr on ShopKpr.Security_Id = SecurityPass.Security_Id) where SecurityPass.Security_Id = S_Id;
           end if;  
        
            
        
       end if;
       end loop;
       close Pcursor;
        
        
  end; 
  |
 
delimiter  ;
